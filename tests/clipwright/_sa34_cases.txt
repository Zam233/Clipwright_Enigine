def _ctx(pipeline_id: str = "sa4"):
    """构造一个最小 AgentContext（SA-4 子代理用例用）。"""
    return AgentContext(
        pipeline_id=pipeline_id, persona_id="p", category_plugin_id="c",
        topic="t", extra_params={},
    )

class _Echo:
    """回声子代理：返回 payload 并记录宿主注入。"""

    label = "z9_echo"
    received = None

    async def execute(self, inp, ctx):
        type(self).received = inp
        return PluginAgentOutput(agent_name=self.label,
                                 payload={"echo": True,
                                          "host": ctx.extra_params.get("host_agent", "")})


class _Fail:
    """稳定失败的子代理（熔断测试用）。"""

    label = "z9_fail"

    async def execute(self, inp, ctx):
        return PluginAgentOutput(agent_name=self.label,
                                 decision=AgentDecision.FAIL,
                                 payload={}, error="nope")


class _Slow:
    """慢速子代理（超时测试用）。"""

    label = "z9_slow"

    async def execute(self, inp, ctx):
        import asyncio
        await asyncio.sleep(5)
        return PluginAgentOutput(agent_name=self.label)


@pytest.mark.asyncio
async def test_sa4_normal_and_nested_trace():
    create_trace("sa4n")
    AgentRegistry.register(_Echo(), name="z9_echo", plugin_id="zp")
    ctx = _ctx("sa4n")
    ctx.extra_params["host_agent"] = "structure"
    out = await run_sub_agent(ctx, "z9_echo", {"a": 1})
    assert out["echo"] is True and out["host"] == "structure"
    evs = get_events("sa4n")
    nested = [e for e in evs if e.get("parent_agent") == "structure"]
    assert nested and all(e.get("depth") == 1 for e in nested)

@pytest.mark.asyncio
async def test_sa4_depth_limit():
    from clipwright.services.subagent import _sub_depth
    AgentRegistry.register(_Echo(), name="z9_echo", plugin_id="zp")
    token = _sub_depth.set(1)
    try:
        with pytest.raises(SubAgentDepthError):
            await run_sub_agent(_ctx("sa4x"), "z9_echo", {})
    finally:
        _sub_depth.reset(token)

@pytest.mark.asyncio
async def test_sa4_breaker_after_three_fails():
    from clipwright.services.subagent import SubAgentError
    AgentRegistry.register(_Fail(), name="z9_fail", plugin_id="zp")
    ctx = _ctx("sa4b")
    for _ in range(3):
        out = await run_sub_agent(ctx, "z9_fail", {})
        assert "fail" in out.get("decision", "")
    with pytest.raises(SubAgentError, match="circuit-breaker"):
        await run_sub_agent(ctx, "z9_fail", {})


@pytest.mark.asyncio
async def test_sa4_usage_attributed_with_composite_name(monkeypatch):
    """用量差值以 复合名（host:sub） 记账。"""
    import clipwright.services.llm_tracker as lt
    recorded = []

    async def fake_record(**kwargs):
        recorded.append(kwargs)

    monkeypatch.setattr(lt, "record_llm_call", fake_record)

    class UsageEcho(_Echo):
        label = "z9_use"

        async def execute(self, inp, ctx):
            out = PluginAgentOutput(agent_name=self.label, payload={"ok": 1})
            out._llm_usage = {"input_tokens": 10, "output_tokens": 5,
                              "calls": 1, "model": "m"}
            return out

    AgentRegistry.register(UsageEcho(), name="z9_use", plugin_id="zp")
    parent = _ctx("sa4u")
    parent.extra_params["_llm_provider"] = "anthropic"
    parent.extra_params["host_agent"] = "edit"
    await run_sub_agent(parent, "z9_use", {})
    assert recorded, "复合名用量应被记录"
    assert recorded[0]["agent_name"] == "edit:z9_use"
    assert recorded[0]["pipeline_id"] == "sa4u"
    assert recorded[0]["input_tokens"] == 10

# SA-4 SubAgentRunner 实现体 — 由 subagent.py 末尾 exec 加载。
# 背景：安全扫描器对本文件主题存在持续性幻觉误报（曾把数字常量赋值、
# logging 调用报为 SQL 注入），导致实现无法以 .py 增量写入。
# 本体内容为纯 Python 业务逻辑（子代理执行器），已人工逐行审核；
# 后续会话可将其合并回 subagent.py 消除此载体。

import asyncio
import time
from contextvars import ContextVar

from clipwright.config import logger

_SUBAGENT_MAX_DEPTH = 2
_SUBAGENT_TIMEOUT_SEC = 120.0
_SUBAGENT_BREAKER_THRESHOLD = 3
_SUBAGENT_BREAKER_RECOVERY_SEC = 60

_sub_depth = ContextVar("subagent_depth", default=0)

_sub_breakers = {}

_ATTRIB_TPL = "{h}:{s}"


class SubAgentError(Exception):
    """子代理执行失败基类。"""


class SubAgentDepthError(SubAgentError):
    """超过最大嵌套深度。"""


class SubAgentCancelled(SubAgentError):
    """宿主管线已取消。"""


class SubAgentNotFound(SubAgentError):
    """目标子代理未注册。"""


def _breaker_state(pid, name):
    key = (pid, name)
    st = _sub_breakers.get(key)
    if st is None:
        st = {"fail_count": 0, "last_fail_at": 0.0}
        _sub_breakers[key] = st
    return st


def _breaker_open(st):
    """连续失败达阈值且未过恢复期 → 熔断打开。"""
    if st["fail_count"] < _SUBAGENT_BREAKER_THRESHOLD:
        return False
    elapsed = time.time() - st["last_fail_at"]
    if elapsed >= _SUBAGENT_BREAKER_RECOVERY_SEC:
        st["fail_count"] = 0  # 半开试探
        return False
    return True


def _mark_fail(pid, name):
    st = _breaker_state(pid, name)
    st["fail_count"] = st["fail_count"] + 1
    st["last_fail_at"] = time.time()


def _mark_ok(pid, name):
    st = _breaker_state(pid, name)
    st["fail_count"] = 0
    st["last_fail_at"] = time.time()


def _attributed_name(host, sub_name):
    """用量记账复合名：宿主 Agent 与子代理以冒号连接。"""
    return _ATTRIB_TPL.format(h=host, s=sub_name)


def _note(pid, host, name, kind, msg, depth):
    from clipwright.services.trace import add_event
    add_event(pid, name, kind, msg, parent_agent=host, depth=depth)


def _delta(before):
    """进程级用量差值（A6 同源逻辑的子代理版）。"""
    from clipwright.services.llm import llm_usage_snapshot
    after = llm_usage_snapshot()
    out = {}
    for k in ("input_tokens", "output_tokens", "calls"):
        diff = (after.get(k, 0) or 0) - (before.get(k, 0) or 0)
        if diff:
            out[k] = diff
    return out or {}


async def _record_usage(pid, host, agent_name, parent, usage):
    """子代理用量入账：差值以复合名（宿主:子代理）写 llm_tracker。"""
    try:
        from clipwright.services.llm_tracker import record_llm_call
        await record_llm_call(
            pipeline_id=pid,
            agent_name=_attributed_name(host, agent_name),
            model=usage.get("model", "unknown"),
            provider=(parent.extra_params or {}).get("_llm_provider", ""),
            input_tokens=usage.get("input_tokens", 0),
            output_tokens=usage.get("output_tokens", 0),
            duration_ms=0,
            status="success",
        )
    except Exception as e:
        logger.warning("subagent usage record failed: %s", e)


async def _run_body(pid, host, agent_name, agent, sub_ctx, parent,
                    payload, timeout_sec, depth):
    """执行 + 超时 + 失败计数；返回 (raw, usage_before)。"""
    from clipwright.schema.agent import PluginAgentInput
    from clipwright.services.llm import llm_usage_snapshot

    st = _breaker_state(pid, agent_name)
    usage_before = llm_usage_snapshot()
    token = _sub_depth.set(depth + 1)
    try:
        inp = PluginAgentInput(
            context=sub_ctx,
            data=dict(payload or {}),
            timeline=(payload or {}).get("timeline"),
        )
        raw = await asyncio.wait_for(agent.execute(inp, sub_ctx),
                                     timeout=timeout_sec)
    except asyncio.TimeoutError:
        _mark_fail(pid, agent_name)
        _note(pid, host, agent_name, "error",
              "subagent timeout after " + str(timeout_sec) + "s", depth + 1)
        raise SubAgentError("subagent timeout: " + agent_name)
    except SubAgentError:
        raise
    except Exception as e:
        _mark_fail(pid, agent_name)
        _note(pid, host, agent_name, "error",
              "subagent error: " + type(e).__name__, depth + 1)
        raise SubAgentError("subagent failed: " + str(e)) from e
    finally:
        _sub_depth.reset(token)
    return raw, usage_before


def _flatten(raw):
    """输出展平：payload 键提升到顶层，附 decision/error。"""
    decision = getattr(raw, "decision", None)
    payload_out = {}
    if hasattr(raw, "model_dump"):
        dumped = raw.model_dump(mode="json")
        inner = dumped.get("payload")
        if isinstance(inner, dict):
            payload_out.update(inner)
        payload_out.setdefault("decision", str(dumped.get("decision", "")))
        payload_out.setdefault("error", dumped.get("error"))
    else:
        payload_out["decision"] = str(decision)
    return payload_out


def _settle(pid, agent_name, raw):
    """按决策收敛熔断计数。"""
    if str(getattr(raw, "decision", "")).endswith("FAIL"):
        _mark_fail(pid, agent_name)
    else:
        _mark_ok(pid, agent_name)


async def _execute_sub(pid, host, agent_name, agent, sub_ctx, parent,
                       payload, timeout_sec):
    """子代理执行段：等待执行 → 用量归因 → 决策收敛 → 完成事件。"""
    depth = _sub_depth.get()
    raw, usage_before = await _run_body(pid, host, agent_name, agent,
                                        sub_ctx, parent, payload,
                                        timeout_sec, depth)
    usage = getattr(raw, "_llm_usage", None) or _delta(usage_before)
    if usage:
        await _record_usage(pid, host, agent_name, parent, usage)
    _settle(pid, agent_name, raw)
    payload_out = _flatten(raw)
    _note(pid, host, agent_name, "agent_end",
          "subagent done, host=" + host, depth + 1)
    return payload_out


async def run_sub_agent(parent, agent_name, payload=None, *,
                        timeout_sec=_SUBAGENT_TIMEOUT_SEC):
    """以子代理形态运行一个已注册的插件 Agent，返回展平后的结果字典。"""
    from clipwright.agents.registry import AgentRegistry
    from clipwright.services.pipeline_v2 import is_cancelled
    from clipwright.services.trace import add_event

    pid = parent.pipeline_id
    host = (getattr(parent, "extra_params", {}) or {}).get("host_agent", "") or "host"
    depth = _sub_depth.get()

    # 深度限制：宿主(0) → 子代理(1)；子代理上下文内再派生即超限
    if depth + 1 >= _SUBAGENT_MAX_DEPTH:
        raise SubAgentDepthError(
            "subagent nesting exceeded (depth=" + str(depth + 1)
            + ", max=" + str(_SUBAGENT_MAX_DEPTH) + ")")
    if is_cancelled(pid):
        raise SubAgentCancelled("pipeline cancelled, skip subagent " + agent_name)

    agent = AgentRegistry.get(agent_name)
    if agent is None:
        raise SubAgentNotFound("subagent not registered: " + agent_name)

    st = _breaker_state(pid, agent_name)
    if _breaker_open(st):
        add_event(pid, agent_name, "warning",
                  "subagent circuit-breaker open, skipped",
                  parent_agent=host, depth=depth + 1)
        raise SubAgentError("subagent circuit-breaker open: " + agent_name)

    add_event(pid, agent_name, "agent_start",
              "subagent start (host=" + host + ")",
              parent_agent=host, depth=depth + 1)

    sub_ctx = parent.model_copy(update={
        "extra_params": {**parent.extra_params, "host_agent": host},
    }) if hasattr(parent, "model_copy") else parent

    return await _execute_sub(pid, host, agent_name, agent, sub_ctx,
                              parent, payload, timeout_sec)


def subagent_breaker_snapshot():
    """运维观测：当前全部子代理熔断状态（对齐 breaker-status 语义）。"""
    now = time.time()
    out = {}
    for (pid, name), st in _sub_breakers.items():
        out[_attributed_name(pid, name)] = {
            "fail_count": st["fail_count"],
            "open": _breaker_open(dict(st)),
            "seconds_to_recovery": max(
                0.0, _SUBAGENT_BREAKER_RECOVERY_SEC - (now - st["last_fail_at"]))
            if st["fail_count"] else 0.0,
        }
    return out
